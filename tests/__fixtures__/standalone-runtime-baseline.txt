      function parseMaskRatio(ratioStr, defaultSteps) {
        var defSteps = (defaultSteps === 4) ? 4 : 2;
        var defBps = (defaultSteps === 4) ? [25, 50, 75, 100] : [50, 100];
        if (!ratioStr || typeof ratioStr !== "string") {
          return { steps: defSteps, breakpoints: defBps };
        }
        var clean = ratioStr.trim();
        if (!clean || (defaultSteps === 4 && (clean === "1:1" || clean === "1"))) return { steps: defSteps, breakpoints: defBps };
        var parts = clean.split(":").map(function(p) { return parseFloat(p.trim()); }).filter(function(n) { return !isNaN(n) && n > 0; });
        if (parts.length === 0) return { steps: defSteps, breakpoints: defBps };
        if (parts.length === 1) {
          var val = Math.min(100, Math.max(1, parts[0]));
          if (val === 100) return { steps: 1, breakpoints: [100] };
          return { steps: 2, breakpoints: [Math.round(val * 100) / 100, 100] };
        }
        var isMonotonic = parts.every(function(val, idx) { return idx === 0 || val > parts[idx - 1]; });
        var maxVal = parts[parts.length - 1];
        if (isMonotonic && (maxVal === 100 || parts.some(function(p) { return p > 10; }))) {
          var bps = parts.map(function(p) { return Math.min(100, Math.round(p * 100) / 100); });
          if (bps[bps.length - 1] < 100) bps.push(100);
          return { steps: bps.length, breakpoints: bps };
        }
        var totalWeight = parts.reduce(function(acc, cur) { return acc + cur; }, 0);
        if (totalWeight <= 0) return { steps: defSteps, breakpoints: defBps };
        var accumulated = 0;
        var bps = [];
        for (var i = 0; i < parts.length; i++) {
          accumulated += parts[i];
          if (i === parts.length - 1) {
            bps.push(100);
          } else {
            bps.push(Math.round((accumulated / totalWeight) * 10000) / 100);
          }
        }
        return { steps: bps.length, breakpoints: bps };
      }

      const INLINE_MD_IMG_RE_STR = "!\\[((?:\\\\.|[^\\]\\n])*)\\]\\(([^()\\n]*(?:\\([^()\\n]*\\)[^()\\n]*)*)\\)(?:[ \\t]*\\{:[^}]*\\})*";

      function cleanInlineKramdownStandalone(text) {
        if (!text || typeof text !== "string") return "";
        const codeTokens = [];
        let safeText = text.replace(/\x60([^\x60]+)\x60/g, function(m) {
          codeTokens.push(m);
          return "%%INLINE_CODE_" + (codeTokens.length - 1) + "%%";
        });
        const imgTokens = [];
        safeText = safeText.replace(new RegExp(INLINE_MD_IMG_RE_STR, "g"), function(_, alt, dest) {
          imgTokens.push("![" + alt + "](" + dest + ")");
          return "%%INLINE_IMG_" + (imgTokens.length - 1) + "%%";
        });
        safeText = safeText.replace(/(<\/[a-zA-Z0-9_-]+>)\s*\{:[^}]*\}/g, "$1");
        const extractStyle = function(attrBody) {
          const m = attrBody.match(/\bstyle\s*=\s*(?:"([^"]*)"|'([^']*)')/i);
          return m ? (m[1] ?? m[2] ?? "").trim() : null;
        };
        const wrapSyntax = function(pattern) {
          safeText = safeText.replace(pattern, function(full, content, attrBody) {
            const styleVal = extractStyle(attrBody);
            return styleVal ? '<span style="' + styleVal + '">' + content + '</span>' : full;
          });
        };
        wrapSyntax(/(\[[^\]]+\]\([^)]+\))\s*\{:\s*([^}]*)\}/g);
        wrapSyntax(/(\*\*[^*]+\*\*)\s*\{:\s*([^}]*)\}/g);
        wrapSyntax(/(__[^_]+__)\s*\{:\s*([^}]*)\}/g);
        wrapSyntax(/(==[^=]+==)\s*\{:\s*([^}]*)\}/g);
        wrapSyntax(/(\*[^*]+\*)\s*\{:\s*([^}]*)\}/g);
        wrapSyntax(/(~~[^~]+~~)\s*\{:\s*([^}]*)\}/g);
        safeText = safeText.replace(/([^\s<>]+)\s*\{:\s*([^}]*)\}/gu, function(full, content, attrBody) {
          const styleVal = extractStyle(attrBody);
          if (!styleVal) return full;
          const leadingDelim = content.match(/^(\*\*|__|==|~~|\*)(.+)$/);
          if (leadingDelim) {
            return leadingDelim[1] + '<span style="' + styleVal + '">' + leadingDelim[2] + '</span>';
          }
          return '<span style="' + styleVal + '">' + content + '</span>';
        });
        safeText = safeText.replace(/\{:[^}]*\}/g, "");
        safeText = safeText.replace(/%%INLINE_IMG_(\d+)%%/g, function(_, idx) {
          return imgTokens[parseInt(idx, 10)] || "";
        });
        safeText = safeText.replace(/%%INLINE_CODE_(\d+)%%/g, function(_, idx) {
          return codeTokens[parseInt(idx, 10)] || "";
        });
        return safeText;
      }

      const SAFE_STANDALONE_INLINE_TAGS = new Set([
        "kbd", "mark", "sub", "sup", "span", "a", "br", "b", "i", "u", "s",
        "del", "em", "strong", "code", "small", "abbr", "ruby", "rt", "rp", "v-mark",

      function formatInlineStandalone(text) {
        if (!text) return "";
        const preprocessed = cleanInlineKramdownStandalone(text);
        const tokens = [];
        const tokenized = preprocessed.replace(/<\/?([a-zA-Z0-9_-]+)(?:\s+[^<>]*)?\/?>/g, function(tagMatch, tagName) {
          const lowerTag = tagName.toLowerCase();
          if (!SAFE_STANDALONE_INLINE_TAGS.has(lowerTag)) return tagMatch;
          if (/on\w+\s*=|javascript:/i.test(tagMatch)) return tagMatch;
          if (lowerTag === "v-mark") {
            const tm = tagMatch.match(/type=["']?([a-zA-Z-]+)["']?/);
            const mt = tm ? tm[1] : "circle";
            tokens.push(tagMatch.startsWith("</") ? "</mark>" : '<mark class="v-mark v-mark--' + mt + '">');
            return "%%SAFE_HTML_" + (tokens.length - 1) + "%%";
          }
          tokens.push(tagMatch);
          return "%%SAFE_HTML_" + (tokens.length - 1) + "%%";
        });
        let res = escapeInlineHtml(tokenized);
        res = res.replace(/\$([^$\n]+)\$/g, '<span class="katex-inline font-math"><em>$1</em></span>');
        res = res.replace(new RegExp(INLINE_MD_IMG_RE_STR, "g"), function(_, alt, rawDest) {
          const url = extractCleanImageUrlStandalone(rawDest);
          const isGif = /\.gif(?:\?|$)/i.test(url);
          return '<img alt="' + alt + '" src="' + url + '" class="' + (isGif ? "slide-inline-img list-item-gif" : "slide-inline-img") + '" />';
        });
        res = res.replace(/\[(.*?)\]\((.*?)\)/g, function(_, title, url) {
          if (url.startsWith("siyuan://blocks/")) {
            const blockId = url.replace("siyuan://blocks/", "").split("?")[0];
            return '<a href="' + url + '" data-ref-id="' + blockId + '" class="slides-ref-tag">🔗 ' + title + '</a>';
          }
          return '<a href="' + url + '" target="_blank" rel="noopener noreferrer" class="slides-link">' + title + '</a>';
        });
        res = res.replace(/\*\*(.*?)\*\*/g, "<strong>$1</strong>");
        res = res.replace(/\*(.*?)\*/g, "<em>$1</em>");
        res = res.replace(/~~(.*?)~~/g, "<del>$1</del>");
        res = res.replace(/\x60([^\x60]+)\x60/g, "<code>$1</code>");
        res = res.replace(/%%SAFE_HTML_(\d+)%%/g, function(_, idxStr) {
          return tokens[parseInt(idxStr, 10)] || "";
        });
        return res;
      }

          function parseRgbLum(rgbStr) {

          function parseHslLum(hslStr) {
            const m = hslStr.match(/hsla?\s*\(\s*([0-9.]+)(?:deg)?\s*[,\s]\s*([0-9.]+)%\s*[,\s]\s*([0-9.]+)%(?:\s*[,\/]\s*([0-9.]+%?))?/i);
            if (!m) return null;

              if (!isNaN(a) && a <= 0.05) return null;

            const h = (((Number(m[1]) % 360) + 360) % 360) / 360;
            const sat = Math.max(0, Math.min(1, Number(m[2]) / 100));
            const lig = Math.max(0, Math.min(1, Number(m[3]) / 100));
            let r, g, b;
            if (sat === 0) {
              r = g = b = lig * 255;
            } else {
              const q = lig < 0.5 ? lig * (1 + sat) : lig + sat - lig * sat;
              const p = 2 * lig - q;
              const hue2rgb = (t) => {
                let tt = t;
                if (tt < 0) tt += 1;
                if (tt > 1) tt -= 1;
                if (tt < 1 / 6) return p + (q - p) * 6 * tt;
                if (tt < 1 / 2) return q;
                if (tt < 2 / 3) return p + (q - p) * (2 / 3 - tt) * 6;
                return p;
              };
              r = hue2rgb(h + 1 / 3) * 255;
              g = hue2rgb(h) * 255;
              b = hue2rgb(h - 1 / 3) * 255;
            }
            return calcLum(r, g, b);

                if (lightNamed.includes(w)) luminances.push(240);
                else if (darkNamed.includes(w)) luminances.push(20);
              }
            }
            if (luminances.length > 0) {
              const avg = luminances.reduce((acc, v) => acc + v, 0) / luminances.length;
              return avg > 140;

            if (/\btransparent\b/.test(s)) return false;

        function syncAutoplayVideos() {
          const doSync = () => {
            const videos = container.querySelectorAll("video");
            videos.forEach((v) => {
              const box = v.closest("[data-autoplay]");
              const isAuto = v.hasAttribute("autoplay") || v.autoplay || (box && box.getAttribute("data-autoplay") === "true");
              if (isAuto) {
                v.muted = true;
                v.defaultMuted = true;
                v.setAttribute("muted", "");
                v.playsInline = true;
                v.setAttribute("playsinline", "");
                const playFn = () => {
                  if (!v.paused) return;
                  v.muted = true;
                  v.defaultMuted = true;
                  const p = v.play();
                  if (p !== undefined) {
                    p.catch(() => {
                      v.muted = true;
                      v.defaultMuted = true;
                      setTimeout(() => { if (v.paused) v.play().catch(() => {}); }, 150);
                    });
                  }
                };
                playFn();
                if (!v.__autoplayBound) {
                  v.__autoplayBound = true;
                  v.addEventListener("loadedmetadata", playFn);
                  v.addEventListener("canplay", playFn);
                }
              }
            });
          };
          setTimeout(doSync, 60);
          setTimeout(doSync, 200);

            const animType = node.getAttribute("data-slides-anim");
            if (animType === "countup") {
              const targetEl = node.querySelector(".slides-countup-target") || node.querySelector("p, h1, h2, h3, h4, h5, h6, span, strong") || node;
              if (!targetEl.getAttribute("data-anim-original")) {
                targetEl.setAttribute("data-anim-original", targetEl.textContent || "");
              }
              const toVal = parseFloat(node.getAttribute("data-countup-to") || "0");
              const fromVal = parseFloat(node.getAttribute("data-countup-from") || "0");
              const durationVal = parseInt(node.getAttribute("data-anim-duration") || "1500", 10);
              const decimalsVal = parseInt(node.getAttribute("data-countup-decimals") || "0", 10);
              const prefixVal = node.getAttribute("data-countup-prefix") || "";
              const suffixVal = node.getAttribute("data-countup-suffix") || "";
              const startTime = performance.now();
              const easeOutExpo = t => (t === 1 ? 1 : 1 - Math.pow(2, -10 * t));
              function update(now) {
                const elapsed = now - startTime;
                const progress = Math.min(1, Math.max(0, elapsed / durationVal));
                const current = fromVal + (toVal - fromVal) * easeOutExpo(progress);
                targetEl.textContent = prefixVal + current.toFixed(decimalsVal) + suffixVal;
                if (progress < 1) {
                  requestAnimationFrame(update);
                }
              }
              targetEl.textContent = prefixVal + fromVal.toFixed(decimalsVal) + suffixVal;
              requestAnimationFrame(update);
            } else if (animType === "typewriter") {
              const targetEl = node.querySelector("p, h1, h2, h3, h4, h5, h6, span, code") || node;
              if (targetEl !== node) {
                node.classList.remove("slides-typewriter-cursor");
                targetEl.classList.add("slides-typewriter-cursor");
              }
              if (!targetEl.getAttribute("data-anim-original")) {
                targetEl.setAttribute("data-anim-original", targetEl.textContent || "");
              }
              const fullText = targetEl.getAttribute("data-anim-original") || targetEl.textContent || "";
              const durationVal = parseInt(node.getAttribute("data-anim-duration") || "1500", 10);
              const stepMs = durationVal / Math.max(fullText.length, 1);
              let idx = 0;
              targetEl.textContent = "";
              function step() {
                idx++;
                targetEl.textContent = fullText.slice(0, idx);
                if (idx < fullText.length) {
                  setTimeout(step, stepMs);
                }
              }
              setTimeout(step, stepMs);
            }
          });
        }
        runSlideAnimations(stage);
      }

      function handleLightboxWheel(e) {
        if (!lightboxContent || lightboxMediaType !== "image") return;
        const img = lightboxContent.querySelector(".slides-lightbox-img");
        if (!img) return;
        e.preventDefault();
        e.stopPropagation();
        if (Math.abs(e.deltaY) < 1) return;

        const factor = e.deltaY < 0 ? 1.2 : 1 / 1.2;
        const currentScale = lightboxScale;
        let nextScale = currentScale * factor;

        if (nextScale <= 1.02) {
          resetLightboxZoom();
          return;
        }
        if (nextScale > 5.0) {
          nextScale = 5.0;
        }

        const rect = img.getBoundingClientRect();
        const currentCenterX = rect.left + rect.width / 2;
        const currentCenterY = rect.top + rect.height / 2;
        const origCenterX = currentCenterX - lightboxTranslateX;
        const origCenterY = currentCenterY - lightboxTranslateY;

        const nextTx = (e.clientX - origCenterX) * (1 - nextScale / currentScale) + lightboxTranslateX * (nextScale / currentScale);
        const nextTy = (e.clientY - origCenterY) * (1 - nextScale / currentScale) + lightboxTranslateY * (nextScale / currentScale);

        lightboxTranslateX = Math.round(nextTx * 10) / 10;
        lightboxTranslateY = Math.round(nextTy * 10) / 10;
        lightboxScale = Math.round(nextScale * 1000) / 1000;
        updateLightboxImgTransform(true);
      }

      if (lightboxContent) {
        lightboxContent.addEventListener("pointerdown", (e) => {
          if (e.button !== 0 || lightboxMediaType !== "image" || lightboxScale <= 1.0) return;
          const img = lightboxContent.querySelector(".slides-lightbox-img");
          if (!img || (e.target !== img && !img.contains(e.target))) return;

          e.stopPropagation();

          lightboxInitialTy = lightboxTranslateY;

            lightboxHasMoved = true;

          lightboxTranslateX = Math.round((lightboxInitialTx + dx) * 10) / 10;
          lightboxTranslateY = Math.round((lightboxInitialTy + dy) * 10) / 10;

          isLightboxDragging = false;

            const origCenterX = currentCenterX - lightboxTranslateX;
            const origCenterY = currentCenterY - lightboxTranslateY;

            lightboxTranslateY = Math.round(((e.clientY - origCenterY) * (1 - targetScale / 1.0)) * 10) / 10;
            lightboxScale = targetScale;
            updateLightboxImgTransform(true);

              canvas.style.backgroundImage = 'url("' + bg + '")';
              canvas.style.backgroundSize = "cover";
              canvas.style.backgroundPosition = "center";
            } else {
              canvas.style.background = bg;
            }
          }
          box.appendChild(canvas);
          const caption = document.createElement("div");
          caption.className = "slides-overview-caption";
          const pageNo = document.createElement("span");
          pageNo.className = "slides-overview-page-no";
          pageNo.textContent = String(idx + 1);
          caption.appendChild(pageNo);
          const title = getSlideOverviewTitle(slide);
          if (title) {
            const titleEl = document.createElement("span");
            titleEl.className = "slides-overview-title";
            titleEl.textContent = title;
            caption.appendChild(titleEl);
          }
          card.appendChild(box);
          card.appendChild(caption);
          card.addEventListener("click", (e) => {
            e.stopPropagation();
            goFromOverview(idx + 1);
          });
          overviewGrid.appendChild(card);
        });
        updateOverviewCurrentMark();
      }

      function renderOverviewThumb(card) {
        if (!card || card.dataset.rendered === "1") return;
        const idx = Number(card.dataset.idx);
        const slide = slides[idx];
        const canvas = card.querySelector(".slides-overview-canvas");
        if (slide && canvas) buildOverviewThumbInto(canvas, slide);
        card.dataset.rendered = "1";
      }

      function ensureOverviewObserver() {
        if (overviewObserver || typeof window.IntersectionObserver !== "function") return;
        overviewObserver = new IntersectionObserver((entries) => {
          for (const en of entries) {
            if (!en.isIntersecting) continue;
            overviewObserver.unobserve(en.target);
            renderOverviewThumb(en.target);
          }
        }, { rootMargin: "240px" });
      }

      function observeOverviewCards() {
        if (!overviewObserver) return;
        const cards = overviewGrid.children;
        for (let i = 0; i < cards.length; i++) {
          if (cards[i].dataset.rendered !== "1") overviewObserver.observe(cards[i]);
        }
      }

      function updateOverviewCurrentMark() {
        const cards = overviewGrid.children;
        for (let i = 0; i < cards.length; i++) {
          const pageNo = cards[i].querySelector(".slides-overview-page-no");
          if (pageNo) pageNo.classList.toggle("is-current", i === currentIndex);
        }
      }

      function setOverviewFocus(page) {
        const total = slides.length || 1;
        overviewFocus = Math.min(Math.max(1, page), total);
        const cards = overviewGrid.children;
        for (let i = 0; i < cards.length; i++) {
          cards[i].classList.toggle("is-active", i + 1 === overviewFocus);
        }
        const el = cards[overviewFocus - 1];
        if (el && el.scrollIntoView) el.scrollIntoView({ block: "nearest", behavior: "smooth" });
      }

      function getOverviewCols() {
        if (!overviewGrid || !overviewGrid.clientWidth) return 1;
        return Math.max(1, Math.round(overviewGrid.clientWidth / (OVERVIEW_CARD_W + 18)));
      }

      function goFromOverview(page) {
        const p = parseInt(page, 10);
        if (isNaN(p) || p < 1 || p > slides.length) return;
        closeOverview();
        renderSlide(p - 1);
        updateOverviewCurrentMark();
      }

      function openOverview() {
        if (slides.length === 0) return;
        buildOverviewCards();
        overviewOpen = true;
        overviewOverlay.style.display = "block";
        ensureOverviewObserver();
        if (overviewObserver) {
          observeOverviewCards();
        } else {
          const cards = overviewGrid.children;
          for (let i = 0; i < cards.length; i++) renderOverviewThumb(cards[i]);
        }
        setOverviewFocus(currentIndex + 1);
      }

      function closeOverview() {
        overviewOpen = false;
        overviewOverlay.style.display = "none";
        hideJumpIndicator();
        if (overviewObserver) {
          const cards = overviewGrid.children;
          for (let i = 0; i < cards.length; i++) overviewObserver.unobserve(cards[i]);
        }
      }

      function toggleOverview() {
        if (overviewOpen) closeOverview();
        else openOverview();
      }

      // 总览打开期间接管全部按键：修复上游总览中空格/PGUP/PGDN 仍会翻页的漏洞
      function handleOverviewKeyDown(e) {
        if (e.ctrlKey || e.metaKey || e.altKey) return;
        if (e.isComposing || isEditableKeyTarget(e.target)) return;
        if (/^[0-9]$/.test(e.key)) {
          if (!jumpInvalid && jumpDisplay.length < 4) {
            jumpDisplay += e.key;
            const p = parseInt(jumpDisplay, 10);
            if (!isNaN(p) && p >= 1 && p <= slides.length) setOverviewFocus(p);
            renderJumpBadge();
            armJumpIdleTimer();
          }
          e.preventDefault();
          return;
        }
        if (e.key === "Enter" && jumpDisplay) {
          if (!jumpInvalid) {
            const page = parseInt(jumpDisplay, 10);
            if (!isNaN(page) && page >= 1 && page <= slides.length) {
              goFromOverview(page);
            } else {
              flashJumpInvalid();
            }
          }
          e.preventDefault();
          return;
        }
        if ((e.key === "Escape" || e.key === "Esc") && (jumpDisplay || jumpInvalid)) {
          hideJumpIndicator();
          e.preventDefault();
          return;
        }
        if (e.key === "ArrowRight") {
          e.preventDefault();
          setOverviewFocus(overviewFocus + 1);
          return;
        }
        if (e.key === "ArrowLeft") {
          e.preventDefault();
          setOverviewFocus(overviewFocus - 1);
          return;
        }
        if (e.key === "ArrowDown" || e.key === "PageDown") {
          e.preventDefault();
          setOverviewFocus(overviewFocus + getOverviewCols());
          return;
        }
        if (e.key === "ArrowUp" || e.key === "PageUp") {
          e.preventDefault();
          setOverviewFocus(overviewFocus - getOverviewCols());
          return;
        }
        if (e.key === " ") {
          e.preventDefault();
          return;
        }
        if (e.key === "Enter") {
          e.preventDefault();
          goFromOverview(overviewFocus);
          return;
        }
        if (e.key === "Escape" || e.key === "Esc" || e.key === "o" || e.key === "O") {
          e.preventDefault();
          closeOverview();
        }
      }

      window.addEventListener("keydown", (e) => {
        if (e.key === "Escape" || e.key === "Esc") {
          if (isLightboxOpen) {
            closeLightbox();
            e.preventDefault();
            return;
          }
        }
        if (isLightboxOpen) return;
        // 总览打开期间全部按键由总览层接管
        if (overviewOpen) {
          handleOverviewKeyDown(e);
          return;
        }
        // 「数字+回车」跳页：可输入元素、修饰键、IME 组合输入中不拦截
        if (!e.ctrlKey && !e.metaKey && !e.altKey && !e.isComposing && !isEditableKeyTarget(e.target)) {
          if (/^[0-9]$/.test(e.key)) {
            if (!jumpInvalid && jumpDisplay.length < 4) {
              jumpDisplay += e.key;
              renderJumpBadge();
              armJumpIdleTimer();
            }
            e.preventDefault();
            return;
          }
          if (e.key === "Enter" && jumpDisplay) {
            if (!jumpInvalid) {
              const page = parseInt(jumpDisplay, 10);
              if (!isNaN(page) && page >= 1 && page <= slides.length) {
                hideJumpIndicator();
                renderSlide(page - 1);
              } else {
                flashJumpInvalid();
              }
            }
            e.preventDefault();
            return;
          }
          if ((e.key === "Escape" || e.key === "Esc") && (jumpDisplay || jumpInvalid)) {
            hideJumpIndicator();
            e.preventDefault();
            return;
          }
          if (e.key === "o" || e.key === "O") {
            e.preventDefault();
            toggleOverview();
            return;
          }
        }
        if (e.key === "ArrowRight" || e.key === " " || e.key === "ArrowDown") goNext();
        if (e.key === "ArrowLeft" || e.key === "ArrowUp") goPrev();
        if (e.key === "f" || e.key === "F") btnFullscreen.click();
      });

      // 监听鼠标滚轮上下翻页调度
      let lastWheelTime = 0;
      const WHEEL_COOLDOWN_MS = 280;
      const WHEEL_DELTA_THRESHOLD = 15;

      function canElementScrollY(el, deltaY) {
        let cur = el;
        while (cur && cur !== document.body && cur !== document.documentElement) {
          if (cur.classList && (cur.classList.contains("slides-viewport") || cur.classList.contains("slides-stage") || cur.classList.contains("slide-container"))) {
            break;
          }
          const style = window.getComputedStyle(cur);
          const overflowY = style.overflowY;
          if (overflowY === "auto" || overflowY === "scroll") {
            const maxScroll = cur.scrollHeight - cur.clientHeight;
            if (maxScroll > 2) {
              if (deltaY > 0 && cur.scrollTop < maxScroll - 1) return true;
              if (deltaY < 0 && cur.scrollTop > 1) return true;
            }
          }
          cur = cur.parentElement;
        }
        return false;
      }

      window.addEventListener("wheel", (e) => {
        if (isLightboxOpen) {
          if (typeof handleLightboxWheel === "function") {
            handleLightboxWheel(e);
          }
          return;
        }
        // 总览打开时滚轮交给总览列表原生滚动，不触发翻页
        if (overviewOpen) return;
        if (Math.abs(e.deltaY) < Math.abs(e.deltaX)) return;
        if (Math.abs(e.deltaY) < WHEEL_DELTA_THRESHOLD) return;

        const target = e.target instanceof Element ? e.target : e.target?.parentElement;
        if (target && canElementScrollY(target, e.deltaY)) return;

        e.preventDefault();
        const now = Date.now();
        if (now - lastWheelTime < WHEEL_COOLDOWN_MS) return;
        lastWheelTime = now;

        if (e.deltaY > 0) {
          goNext();
        } else {
          goPrev();
        }
      }, { passive: false });

      document.addEventListener("click", (e) => {
        if (isLightboxOpen) {
          if (typeof lightboxHasMoved !== "undefined" && lightboxHasMoved) {
            lightboxHasMoved = false;
            e.stopPropagation();
            return;
          }
          const target = e.target instanceof Element ? e.target : e.target?.parentElement;
          const isMediaItself = target && target.closest && target.closest(".slides-lightbox-img, .slides-lightbox-video");
          if (!isMediaItself) {
            e.stopPropagation();
            closeLightbox();
          }
          return;
        }
        // 总览打开时空白点击由总览层自身处理，不触发底层翻页
        if (overviewOpen) return;
        if (e.button !== 0) return;
        const targetEl = e.target instanceof Element ? e.target : e.target?.parentElement;
        if (!targetEl || !targetEl.closest) return;

        const hasPendingMask = currentMaskConfig && (currentMaskStep < currentMaskConfig.totalSteps);
        if (hasPendingMask) {
          const isBottomControls = targetEl.closest(".controls-bar, .controls, .slides-toc");
          if (isBottomControls) return;
          e.preventDefault();
          e.stopPropagation();
          goNext();
          return;
        }

        const refEl = targetEl.closest("[data-ref-id]");
        if (refEl) {
          const id = refEl.getAttribute("data-ref-id");
          if (id) {
            window.open("siyuan://blocks/" + id, "_blank");
            return;
          }
        }

        // 媒体点击唤起 Lightbox 放大模态
        const mediaEl = targetEl.closest(".slide-content img, .slide-content video");
        if (mediaEl && !mediaEl.closest(".controls, .controls-bar, button, a")) {
          e.stopPropagation();
          const isVideo = mediaEl.tagName.toLowerCase() === "video";
          const src = isVideo ? (mediaEl.currentSrc || mediaEl.src || mediaEl.querySelector("source")?.src) : mediaEl.src;
          if (src) {
            openLightbox(isVideo ? "video" : "image", src, mediaEl.alt || "");
            return;
          }
        }

        const selection = window.getSelection ? window.getSelection().toString().trim() : "";
        if (selection.length > 0) return;

        const isComponent = targetEl.closest(
          ".slide-block, .controls-bar, .image-right-img-container, .image-left-img-container, .controls, img, video, audio, button, a, input, textarea, select, [data-clickable], [data-ref-id], .slides-toc, .empty-state"
        );
        if (isComponent) return;

        const isLeft = e.clientX < window.innerWidth / 2;
        if (isLeft) {
          goPrev();
        } else {
          goNext();
        }
      });

      renderSlide(0);
      updateScale();
    })();
